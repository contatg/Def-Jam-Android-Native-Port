cmake_minimum_required(VERSION 3.20)
project(kernel_irql_abi_test C)
if(NOT WIN32)
    message(FATAL_ERROR "This synthetic-memory regression uses Windows VirtualAlloc")
endif()
set(CMAKE_C_STANDARD 11)
add_subdirectory(../.. runtime)
add_executable(kernel_irql_abi_test test_main.c)
target_include_directories(kernel_irql_abi_test PRIVATE ../../src/kernel)
target_link_libraries(kernel_irql_abi_test PRIVATE xboxrecomp)
enable_testing()
add_test(NAME kernel_irql_abi_test COMMAND kernel_irql_abi_test)
# Synthetic low RAM only; no GPU register aperture is mapped.
set_tests_properties(kernel_irql_abi_test PROPERTIES ENVIRONMENT "RECOMP_GPU_PREEMPT=0")
