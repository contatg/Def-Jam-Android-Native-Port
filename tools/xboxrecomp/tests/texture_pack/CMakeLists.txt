cmake_minimum_required(VERSION 3.20)
project(texture_pack_fixture C)
set(CMAKE_C_STANDARD 11)
add_executable(texture_pack_test test_main.c)
target_include_directories(texture_pack_test PRIVATE ../../src)
if(WIN32)
    target_compile_definitions(texture_pack_test PRIVATE _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
# Assertions are the fixture's checks and include setup calls; retain in Release.
if(MSVC)
    target_compile_options(texture_pack_test PRIVATE /UNDEBUG)
else()
    target_compile_options(texture_pack_test PRIVATE -UNDEBUG)
endif()
if(WIN32)
    target_link_libraries(texture_pack_test PRIVATE windowscodecs ole32 bcrypt shell32)
else()
    target_include_directories(texture_pack_test PRIVATE ../../src/platform ../../src/d3d)
    find_package(Threads REQUIRED)    # the background decoder
    target_link_libraries(texture_pack_test PRIVATE Threads::Threads)
endif()
enable_testing()
add_test(NAME texture_pack COMMAND texture_pack_test)

if(NOT WIN32)
    # The portable SHA-256 (Linux) is also tested where the system one is used.
    add_executable(texture_pack_portable_sha test_main.c)
    target_include_directories(texture_pack_portable_sha PRIVATE ../../src ../../src/platform ../../src/d3d)
    target_compile_definitions(texture_pack_portable_sha PRIVATE TEXTURE_PACK_PORTABLE_SHA=1)
    target_compile_options(texture_pack_portable_sha PRIVATE -UNDEBUG)
    target_link_libraries(texture_pack_portable_sha PRIVATE Threads::Threads)
    add_test(NAME texture_pack_portable_sha COMMAND texture_pack_portable_sha)
    # The C rounds alone, where the x86 SHA extensions would otherwise be used.
    add_executable(texture_pack_c_sha test_main.c)
    target_include_directories(texture_pack_c_sha PRIVATE ../../src ../../src/platform ../../src/d3d)
    target_compile_definitions(texture_pack_c_sha PRIVATE TEXTURE_PACK_PORTABLE_SHA=1 TEXTURE_PACK_NO_SHA_NI=1)
    target_compile_options(texture_pack_c_sha PRIVATE -UNDEBUG)
    target_link_libraries(texture_pack_c_sha PRIVATE Threads::Threads)
    add_test(NAME texture_pack_c_sha COMMAND texture_pack_c_sha)
endif()
