cmake_minimum_required(VERSION 3.20)
add_library(defjam_fg34 SHARED bridge34.cpp vendor/src/framegen.cpp)
target_compile_features(defjam_fg34 PRIVATE cxx_std_17)
target_compile_definitions(defjam_fg34 PRIVATE VK_NO_PROTOTYPES=1)
target_include_directories(defjam_fg34 PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${CMAKE_CURRENT_LIST_DIR}/vendor/src")
set_target_properties(defjam_fg34 PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_compile_options(defjam_fg34 PRIVATE -O2 -ffunction-sections -fdata-sections)
target_link_options(defjam_fg34 PRIVATE -Wl,--gc-sections "-Wl,--version-script=${CMAKE_CURRENT_LIST_DIR}/exports.map")
if(ANDROID)
    target_link_libraries(defjam_fg34 PRIVATE log)
endif()
